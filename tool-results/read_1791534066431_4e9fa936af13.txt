   293→// PURCHASE RECOVERY — full width, clear context
   294→// ════════════════════════════════════════════════════════════════════════
   295→function PurchaseRecoveryCard({ sales, purchaseCost, cur }: { sales: number; purchaseCost: number; cur: string }) {
   296→  if (purchaseCost <= 0) {
   297→    return (
   298→      <Card className="card-raised bg-card p-3.5">
   299→        <p className="text-[11px] font-semibold uppercase tracking-wide text-muted-foreground">Purchase Recovery</p>
   300→        <p className="mt-1.5 text-xl font-bold tnum text-muted-foreground">—</p>
   301→        <p className="text-[11px] text-muted-foreground">No purchases today</p>
   302→      </Card>
   303→    );
   304→  }
   305→  const pct = (sales / purchaseCost) * 100;
   306→  const diff = sales - purchaseCost;
   307→  const isBreakEven = Math.abs(diff) < 0.005;
   308→  const isAbove = diff > 0;
   309→  const color = isBreakEven ? "#2563EB" : isAbove ? "#16803C" : "#C53030";
   310→  return (
   311→    <Card className="card-raised bg-card p-3.5">
   312→      <div className="flex items-end justify-between">
   313→        <div>
   314→          <p className="text-[11px] font-semibold uppercase tracking-wide text-muted-foreground">Purchase Recovery</p>
   315→          <p className="mt-0.5 text-xl font-bold tnum leading-none" style={{ color }}>{pct.toFixed(1)}%</p>
   316→          <p className="mt-0.5 text-[11px] text-muted-foreground">of purchase cost recovered by sales</p>
   317→        </div>
   318→        <div className="text-right">
   319→          <p className="text-[11px] text-muted-foreground tnum">{formatBirr(sales, cur)} in</p>
   320→          <p className="text-[11px] text-muted-foreground tnum">{formatBirr(purchaseCost, cur)} spent</p>
   321→        </div>
   322→      </div>
   323→      <div className="relative mt-2.5 h-2 overflow-hidden rounded-full bg-muted/60">
   324→        <div className="h-full rounded-full" style={{ width: `${Math.min(100, (pct / 200) * 100)}%`, backgroundColor: color }} />
   325→        <div className="absolute left-1/2 top-0 h-full w-px bg-foreground/40" />
   326→      </div>
   327→      <p className="mt-2 text-[11px] font-medium" style={{ color }}>
   328→        {isBreakEven
   329→          ? "✓ Purchase cost fully recovered"
   330→          : isAbove
   331→          ? `+${formatBirr(diff, cur)} above purchase cost`
   332→          : `−${formatBirr(Math.abs(diff), cur)} still needed to cover purchases`}
   333→      </p>
   334→    </Card>
   335→  );
   336→}

   339→// NET RESULT — full width, clear context
   340→// ════════════════════════════════════════════════════════════════════════
   341→function NetResultCard({ sales, purchaseCost, expenses, cur }: { sales: number; purchaseCost: number; expenses: number; cur: string }) {
   342→  const net = Math.round((sales - purchaseCost - expenses) * 100) / 100;
   343→  const isZero = Math.abs(net) < 0.005;
   344→  const isPositive = net > 0;
   345→  const color = isZero ? "#2563EB" : isPositive ? "#16803C" : "#C53030";
   346→  const label = isZero ? "Break Even" : isPositive ? "Profit" : "Loss";
   347→  return (
   348→    <Card className="card-raised bg-card p-3.5">
   349→      <div className="flex items-end justify-between">
   350→        <div>
   351→          <p className="text-[11px] font-semibold uppercase tracking-wide text-muted-foreground">Net Result</p>
   352→          <p className="mt-0.5 text-xl font-bold tnum leading-none" style={{ color }}>
   353→            {net < 0 ? "−" : isPositive ? "+" : ""}{formatBirr(Math.abs(net), cur)}
   354→          </p>
   355→          <p className="mt-0.5 text-[11px] font-medium" style={{ color }}>{label} today</p>
   356→        </div>
   357→        <div className="text-right space-y-0.5 text-[11px] tnum text-muted-foreground">
   358→          <p>+ {formatBirr(sales, cur)} <span className="text-muted-foreground/60">sales</span></p>
   359→          <p>− {formatBirr(purchaseCost, cur)} <span className="text-muted-foreground/60">purchase</span></p>
   360→          <p>− {formatBirr(expenses, cur)} <span className="text-muted-foreground/60">expenses</span></p>
   361→        </div>
   362→      </div>
   363→    </Card>
   364→  );
   365→}

   368→// STOCK FLOW — single unified card: Bought → Sold → Remaining
   369→// No duplicate numbers. One clear picture of today's meat flow.
   370→// ════════════════════════════════════════════════════════════════════════
   371→function StockFlowCard({ kgSold, kgBought, cur: _cur }: { kgSold: number; kgBought: number; cur: string }) {
   372→  const remaining = Math.max(0, Math.round((kgBought - kgSold) * 100) / 100);
   373→  const soldPct = kgBought > 0 ? Math.round((kgSold / kgBought) * 100) : 0;
   374→  const barColor = soldPct >= 50 ? "bg-emerald-500" : soldPct > 0 ? "bg-amber-500" : "bg-muted-foreground/40";
   375→
   376→  return (
   377→    <Card className="mb-3 card-raised bg-card p-3">
   378→      <div className="mb-2 flex items-center justify-between">
   379→        <p className="text-[10px] font-semibold uppercase tracking-wide text-muted-foreground">Stock Flow Today</p>
   380→        <span className="text-[10px] font-bold tnum text-muted-foreground">{soldPct}% sold</span>
   381→      </div>
   382→      {/* Three-column flow: Bought → Sold → Left */}
   383→      <div className="grid grid-cols-3 gap-2 text-center">
   384→        <div className="rounded-lg bg-amber-500/8 px-2 py-2">
   385→          <p className="text-[9px] font-semibold uppercase tracking-wide text-amber-400">Bought</p>
   386→          <p className="mt-0.5 text-sm font-bold tnum">{formatKg(kgBought)}</p>
   387→        </div>
   388→        <div className="rounded-lg bg-emerald-500/8 px-2 py-2">
   389→          <p className="text-[9px] font-semibold uppercase tracking-wide text-emerald-400">Sold</p>
   390→          <p className="mt-0.5 text-sm font-bold tnum">{formatKg(kgSold)}</p>
   391→        </div>

   395→        </div>
   396→      </div>
   397→      {/* Single progress bar — sold portion of bought */}
   398→      {kgBought > 0 && (

   400→          <div className={`h-full rounded-full ${barColor}`} style={{ width: `${Math.min(100, soldPct)}%` }} />
