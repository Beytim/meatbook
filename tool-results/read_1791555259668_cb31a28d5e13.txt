   296→// PURCHASE RECOVERY — full width, clear context
   297→// ════════════════════════════════════════════════════════════════════════
   298→function PurchaseRecoveryCard({ sales, purchaseCost, cur, t }: { sales: number; purchaseCost: number; cur: string; t: (k: string) => string }) {
   299→  if (purchaseCost <= 0) {
   300→    return (
   301→      <Card className="card-raised bg-card p-3.5">
   302→        <p className="text-[11px] font-semibold uppercase tracking-wide text-muted-foreground">{t("home.purchaseRecovery")}</p>
   303→        <p className="mt-1.5 text-xl font-bold tnum text-muted-foreground">—</p>
   304→        <p className="text-[11px] text-muted-foreground">{t("home.noPurchasesToday")}</p>
   305→      </Card>
   306→    );
   307→  }
   308→  const pct = (sales / purchaseCost) * 100;
   309→  const diff = sales - purchaseCost;
   310→  const isBreakEven = Math.abs(diff) < 0.005;
   311→  const isAbove = diff > 0;
   312→  const color = isBreakEven ? "#2563EB" : isAbove ? "#16803C" : "#C53030";
   313→  return (
   314→    <Card className="card-raised bg-card p-3.5">
   315→      <div className="flex items-end justify-between">
   316→        <div>
   317→          <p className="text-[11px] font-semibold uppercase tracking-wide text-muted-foreground">{t("home.purchaseRecovery")}</p>
   318→          <p className="mt-0.5 text-xl font-bold tnum leading-none" style={{ color }}>{pct.toFixed(1)}%</p>
   319→          <p className="mt-0.5 text-[11px] text-muted-foreground">{t("home.ofPurchaseCostRecovered")}</p>
   320→        </div>
   321→        <div className="text-right">
   322→          <p className="text-[11px] text-muted-foreground tnum">{formatBirr(sales, cur)} in</p>
   323→          <p className="text-[11px] text-muted-foreground tnum">{formatBirr(purchaseCost, cur)} spent</p>
   324→        </div>
   325→      </div>
   326→      <div className="relative mt-2.5 h-2 overflow-hidden rounded-full bg-muted/60">
   327→        <div className="h-full rounded-full" style={{ width: `${Math.min(100, (pct / 200) * 100)}%`, backgroundColor: color }} />
   328→        <div className="absolute left-1/2 top-0 h-full w-px bg-foreground/40" />
   329→      </div>
   330→      <p className="mt-2 text-[11px] font-medium" style={{ color }}>
   331→        {isBreakEven
   332→          ? t("home.costRecovered")
   333→          : isAbove
   334→          ? `+${formatBirr(diff, cur)} ${t("home.abovePurchaseCost")}`
   335→          : `−${formatBirr(Math.abs(diff), cur)} ${t("home.stillNeeded")}`}
   336→      </p>
   337→    </Card>
   338→  );
   339→}

   344→function NetResultCard({ sales, purchaseCost, expenses, cur, t }: { sales: number; purchaseCost: number; expenses: number; cur: string; t: (k: string) => string }) {
   345→  const net = Math.round((sales - purchaseCost - expenses) * 100) / 100;
   346→  const isZero = Math.abs(net) < 0.005;
   347→  const isPositive = net > 0;
   348→  const color = isZero ? "#2563EB" : isPositive ? "#16803C" : "#C53030";
   349→  const label = isZero ? t("home.breakEvenLabel") : isPositive ? t("home.profit") : t("home.loss");
   350→  return (
   351→    <Card className="card-raised bg-card p-3.5">
   352→      <div className="flex items-end justify-between">
   353→        <div>
   354→          <p className="text-[11px] font-semibold uppercase tracking-wide text-muted-foreground">{t("home.netResult")}</p>
   355→          <p className="mt-0.5 text-xl font-bold tnum leading-none" style={{ color }}>
   356→            {net < 0 ? "−" : isPositive ? "+" : ""}{formatBirr(Math.abs(net), cur)}
   357→          </p>
   358→          <p className="mt-0.5 text-[11px] font-medium" style={{ color }}>{label} {t("home.todayLabel")}</p>
   359→        </div>
   360→        <div className="text-right space-y-0.5 text-[11px] tnum text-muted-foreground">
