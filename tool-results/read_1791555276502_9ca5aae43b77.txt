    24→// ─── Types ──────────────────────────────────────────────────────────────
    25→interface CashSession {
    26→  id: string;
    27→  opening: number;
    28→  expected: number | null;
    29→  counted: number | null;
    30→  difference: number | null;
    31→  status: "OPEN" | "CLOSED";
    32→  openedAt: string;
    33→  closedAt: string | null;
    34→  openedBy: string | null;
    35→  closedBy: string | null;
    36→  note: string | null;
    37→}
    38→interface SessionsResp {
    39→  sessions: CashSession[];
    40→  open: (CashSession & { expected: number }) | null;
    41→}
    42→
    43→async function fetchSessions(): Promise<SessionsResp> {
    44→  const r = await fetch("/api/meat/cash/sessions");
    45→  if (!r.ok) throw new Error("failed");
    46→  return r.json();
    47→}

   299→// ─── Open Drawer Dialog ─────────────────────────────────────────────────
   300→function OpenDrawerDialog({
   301→  open, onOpenChange, onSuccess,
   302→}: {
   303→  open: boolean;
   304→  onOpenChange: (o: boolean) => void;
   305→  onSuccess: () => void;
   306→}) {
   307→  const { t } = useLang();
   308→  const [opening, setOpening] = React.useState("");
   309→  const [userName, setUserName] = React.useState("");
   310→
   311→  const canOpen = !!opening && (parseFloat(opening) || 0) >= 0;
   312→
   313→  const mut = useMutation({
   314→    mutationFn: async () => {
   315→      const r = await fetch("/api/meat/cash/sessions", {
   316→        method: "POST",
   317→        headers: { "Content-Type": "application/json" },
   318→        body: JSON.stringify({
   319→          action: "OPEN",
   320→          opening: Number(opening) || 0,
   321→          userName: userName.trim() || undefined,
   322→        }),
   323→      });
   324→      if (!r.ok) {
   325→        const j = await r.json().catch(() => null);
   326→        throw new Error(j?.error || "failed");
   327→      }
   328→      return r.json();
   329→    },
   330→    onSuccess: () => {
   331→      toast.success(t("cashControl.openedToast"));
   332→      setOpening("");
   333→      setUserName("");
   334→      onSuccess();
   335→    },
   336→    onError: (e: Error) => toast.error(e.message || t("saleFailed")),
   337→  });
   338→
   339→  return (
   340→    <Dialog open={open} onOpenChange={onOpenChange}>
   341→      <DialogContent className="max-w-md">
   342→        <DialogHeader>
   343→          <DialogTitle>{t("cashControl.openDrawer")}</DialogTitle>
   344→        </DialogHeader>
   345→        <div className="space-y-3 py-2">
   346→          <div>
   347→            <Label className="text-xs">{t("cashControl.openAmount")} (Br)</Label>
   348→            <Input
   349→              value={opening}
   350→              onChange={(e) => setOpening(e.target.value)}
   351→              inputMode="decimal"
   352→              placeholder="0.00"
   353→              className="mt-1 tnum text-lg"
   354→              autoFocus
   355→            />
   356→            <p className="mt-1 text-[11px] text-muted-foreground">
   357→              {t("cashControl.subtitle")}
   358→            </p>
   359→          </div>
   360→          <div>
   361→            <Label className="text-xs">{t("cashControl.openedBy")}</Label>
   362→            <Input
   363→              value={userName}
   364→              onChange={(e) => setUserName(e.target.value)}
   365→              placeholder="…"
   366→              className="mt-1"
   367→            />
   368→          </div>
   369→        </div>
   370→        <DialogFooter>
   371→          <Button variant="outline" onClick={() => onOpenChange(false)}>{t("common.cancel")}</Button>
   372→          <Button
   373→            onClick={() => mut.mutate()}
   374→            disabled={mut.isPending || !canOpen}
   375→            className="bg-primary text-primary-foreground"
   376→          >
   377→            {mut.isPending ? t("common.loading") : t("cashControl.openDrawer")}
   378→          </Button>
   379→        </DialogFooter>
   380→      </DialogContent>
   381→    </Dialog>
   382→  );
   383→}

   385→// ─── Close Drawer Dialog ────────────────────────────────────────────────
   386→function CloseDrawerDialog({
   387→  open, onOpenChange, expected, onSuccess,
   388→}: {
   389→  open: boolean;
   390→  onOpenChange: (o: boolean) => void;
   391→  expected: number;
   392→  onSuccess: (session: CashSession) => void;
   393→}) {
   394→  const { t } = useLang();
   395→  const qc = useQueryClient();
   396→  const [counted, setCounted] = React.useState("");
   397→  const [note, setNote] = React.useState("");
   398→  const [userName, setUserName] = React.useState("");
   399→  const [transferEnabled, setTransferEnabled] = React.useState(false);
   400→  const [transferAmount, setTransferAmount] = React.useState("");
   401→  const [transferTo, setTransferTo] = React.useState<"MOBILE" | "BANK">("BANK");
   402→  const [transferDetail, setTransferDetail] = React.useState("");
   403→
   404→  const countedN = parseFloat(counted) || 0;
   405→  const diff = counted ? countedN - expected : 0;
   406→  const transferN = parseFloat(transferAmount) || 0;
   407→
   408→  const mut = useMutation({
   409→    mutationFn: async () => {
   410→      // If transfer is enabled, create a money move (OUT from CASH, IN to selected account)
   411→      if (transferEnabled && transferN > 0) {
   412→        await fetch("/api/meat/money", {
   413→          method: "POST",
   414→          headers: { "Content-Type": "application/json" },
   415→          body: JSON.stringify({
   416→            direction: "OUT", account: "CASH", amount: transferN,
   417→            reason: `Cash drawer transfer to ${transferTo}`,
   418→            userName: userName.trim() || "Abebe Owner",
   419→          }),
   420→        });
   421→        await fetch("/api/meat/money", {
   422→          method: "POST",
   423→          headers: { "Content-Type": "application/json" },
   424→          body: JSON.stringify({
   425→            direction: "IN", account: transferTo, detail: transferDetail || undefined,
   426→            amount: transferN,
   427→            reason: `Cash drawer transfer`,
   428→            userName: userName.trim() || "Abebe Owner",
   429→          }),
   430→        });
   431→      }
   432→      // Build note with transfer info if applicable
   433→      const transferNote = transferEnabled && transferN > 0
   434→        ? `Transferred ${formatBirr(transferN)} to ${transferTo}${transferDetail ? ` · ${transferDetail}` : ""}`
   435→        : "";
   436→      const fullNote = [note.trim(), transferNote].filter(Boolean).join(" | ") || undefined;
   437→      // Close the drawer
   438→      const r = await fetch("/api/meat/cash/sessions", {
   439→        method: "POST",
   440→        headers: { "Content-Type": "application/json" },
   441→        body: JSON.stringify({
   442→          action: "CLOSE",
   443→          counted: countedN,
   444→          note: fullNote,
   445→          userName: userName.trim() || undefined,
   446→        }),
   447→      });
   448→      if (!r.ok) {
   449→        const j = await r.json().catch(() => null);
   450→        throw new Error(j?.error || "failed");
   451→      }
   452→      return r.json();
   453→    },
   454→    onSuccess: (data) => {
   455→      toast.success(t("cashControl.closedToast"));
   456→      if (transferEnabled && transferN > 0) {
   457→        toast.success(`${formatBirr(transferN)} → ${transferTo}`);
   458→      }
   459→      qc.invalidateQueries({ queryKey: ["dashboard"] });
   460→      qc.invalidateQueries({ queryKey: ["money"] });
   461→      qc.invalidateQueries({ queryKey: ["cash-sessions"] });
   462→      const session: CashSession = data.session;
   463→      setCounted(""); setNote(""); setUserName("");
   464→      setTransferEnabled(false); setTransferAmount(""); setTransferDetail("");
   465→      onSuccess(session);
   466→    },
   467→    onError: (e: Error) => toast.error(e.message || t("saleFailed")),
   468→  });
   469→
   470→  return (
   471→    <Dialog open={open} onOpenChange={onOpenChange}>
   472→      <DialogContent className="max-h-[90vh] max-w-md overflow-y-auto">
   473→        <DialogHeader>
   474→          <DialogTitle>{t("cashControl.closeDrawer")}</DialogTitle>
   475→          <DialogDescription className="sr-only">{t("cashControl.subtitle")}</DialogDescription>
   476→        </DialogHeader>
   477→        <div className="space-y-3 py-2">
   478→          {/* Expected */}
   479→          <div className="flex items-center justify-between rounded-lg bg-muted/40 px-3 py-2.5">
   480→            <span className="text-xs font-medium text-muted-foreground">{t("cashControl.expected")}</span>
   481→            <span className="text-base font-bold tnum">{formatBirr(expected)}</span>
   482→          </div>
   483→
   484→          <div>
   485→            <Label className="text-xs">{t("cashControl.countedAmount")} (Br)</Label>
   486→            <Input
   487→              value={counted}
   488→              onChange={(e) => setCounted(e.target.value)}
   489→              inputMode="decimal"
   490→              placeholder="0.00"
   491→              className="mt-1 tnum text-lg"
   492→              autoFocus
   493→            />
   494→          </div>
   495→
   496→          {/* Difference live */}
   497→          {counted ? (
   498→            <div className={cn(
   499→              "flex items-center justify-between rounded-lg px-3 py-2.5 ring-1",
   500→              diff === 0
   501→                ? "bg-emerald-500/10 ring-emerald-500/25"
   502→                : diff > 0
   503→                ? "bg-emerald-500/10 ring-emerald-500/25"
   504→                : "bg-red-500/10 ring-red-500/25",
   505→            )}>
   506→              <span className="text-xs font-medium text-muted-foreground">{t("cashControl.difference")}</span>
   507→              <Pill tone={diffTone(diff)}>{diffLabel(diff)}</Pill>
   508→            </div>
   509→          ) : (
   510→            <div className="flex items-center justify-between rounded-lg bg-muted/30 px-3 py-2.5">
   511→              <span className="text-xs font-medium text-muted-foreground">{t("cashControl.difference")}</span>
   512→              <span className="text-sm font-semibold text-muted-foreground tnum">—</span>
   513→            </div>
   514→          )}
   515→
   516→          {/* Transfer option */}
   517→          <div className="rounded-lg border border-border/60 p-2.5">
   518→            <label className="flex items-center gap-2">
   519→              <input type="checkbox" checked={transferEnabled} onChange={(e) => setTransferEnabled(e.target.checked)} className="h-4 w-4 rounded accent-primary" />
   520→              <span className="text-xs font-semibold">{t("money.bank")} / {t("money.mobile")}</span>
   521→            </label>
   522→            {transferEnabled && (
   523→              <div className="mt-2.5 space-y-2">
   524→                <div className="grid grid-cols-2 gap-1.5">
   525→                  <button type="button" onClick={() => { setTransferTo("BANK"); setTransferDetail(""); }}
   526→                    className={cn("rounded-lg py-2 text-xs font-semibold", transferTo === "BANK" ? "bg-primary text-primary-foreground" : "bg-muted/60 text-muted-foreground")}>
   527→                    🏦 {t("money.bank")}
   528→                  </button>
   529→                  <button type="button" onClick={() => { setTransferTo("MOBILE"); setTransferDetail(""); }}
   530→                    className={cn("rounded-lg py-2 text-xs font-semibold", transferTo === "MOBILE" ? "bg-primary text-primary-foreground" : "bg-muted/60 text-muted-foreground")}>
   531→                    📱 {t("money.mobile")}
   532→                  </button>
   533→                </div>
   534→                <div>
   535→                  <Label className="text-[10px]">{t("money.amount")} (Br)</Label>
   536→                  <Input value={transferAmount} onChange={(e) => setTransferAmount(e.target.value)} inputMode="decimal" placeholder="0.00" className="mt-0.5 h-9 tnum text-sm" />
   537→                </div>
   538→                <div>
   539→                  <Label className="text-[10px]">{transferTo === "MOBILE" ? t("sell.provider") : t("sell.bankLabel")}</Label>
   540→                  <div className="mt-0.5">
   541→                    <AccountProviderSelect method={transferTo} value={transferDetail} onChange={setTransferDetail} />
   542→                  </div>
   543→                </div>
   544→              </div>
   545→            )}
   546→          </div>
   547→
   548→          <div>
   549→            <Label className="text-xs">{t("cashControl.closedBy")}</Label>
   550→            <Input value={userName} onChange={(e) => setUserName(e.target.value)} placeholder="…" className="mt-1" />
   551→          </div>
   552→
   553→          <div>
   554→            <Label className="text-xs">{t("cashControl.note")}</Label>
   555→            <Textarea value={note} onChange={(e) => setNote(e.target.value)} rows={2} placeholder="…" className="mt-1" />
   556→          </div>
   557→        </div>
   558→
   559→        <DialogFooter>
   560→          <Button variant="outline" onClick={() => onOpenChange(false)}>{t("common.cancel")}</Button>
   561→          <Button
   562→            onClick={() => mut.mutate()}
   563→            disabled={mut.isPending || !counted}
   564→            className="bg-primary text-primary-foreground"
   565→          >
   566→            {mut.isPending ? t("common.loading") : t("cashControl.closeDrawer")}
   567→          </Button>
   568→        </DialogFooter>
   569→      </DialogContent>
   570→    </Dialog>
   571→  );
   572→}

   574→// ─── Helpers ────────────────────────────────────────────────────────────
   575→function diffTone(diff: number): "good" | "bad" {
   576→  if (diff === 0) return "good";
   577→  return diff > 0 ? "good" : "bad";
   578→}
   579→
   580→function diffAccent(diff: number): "emerald" | "red" | undefined {
   581→  if (diff === 0) return "emerald";
   582→  return diff > 0 ? "emerald" : "red";
   583→}
   584→
   585→function diffLabel(diff: number): string {
   586→  if (diff === 0) return "";
   587→  if (diff > 0) return `+${formatBirr(diff)}`;
   588→  return `−${formatBirr(Math.abs(diff))}`;
   589→}

   591→// ─── Small display pieces ───────────────────────────────────────────────
   592→function MetaCell({
   593→  label, value, tnum, tone, accent,
   594→}: {
   595→  label: string;
   596→  value: string;
   597→  tnum?: boolean;
   598→  tone?: "good" | "bad";
   599→  accent?: "emerald" | "red";
   600→}) {
   601→  const valueCls = tone === "good"
   602→    ? "text-emerald-400"
   603→    : tone === "bad"
   604→    ? "text-red-400"
   605→    : accent === "emerald"
   606→    ? "text-emerald-400"
   607→    : accent === "red"
   608→    ? "text-red-400"
   609→    : "text-foreground";
   610→  return (
   611→    <div className="rounded-lg bg-muted/40 px-2.5 py-2">
   612→      <p className="text-[10px] uppercase tracking-wider text-muted-foreground">{label}</p>
   613→      <p className={cn("truncate text-xs font-medium", valueCls, tnum && "tnum")}>{value}</p>
   614→    </div>
   615→  );
   616→}

   638→        isAccent && "text-muted-foreground",

   641→      </p>
   642→    </div>
   643→  );
