    23→// ─── Types ──────────────────────────────────────────────────────────────
    24→interface CashSession {
    25→  id: string;
    26→  opening: number;
    27→  expected: number | null;
    28→  counted: number | null;
    29→  difference: number | null;
    30→  status: "OPEN" | "CLOSED";
    31→  openedAt: string;
    32→  closedAt: string | null;
    33→  openedBy: string | null;
    34→  closedBy: string | null;
    35→  note: string | null;
    36→}
    37→interface SessionsResp {
    38→  sessions: CashSession[];
    39→  open: (CashSession & { expected: number }) | null;
    40→}
    41→
    42→async function fetchSessions(): Promise<SessionsResp> {
    43→  const r = await fetch("/api/meat/cash/sessions");
    44→  if (!r.ok) throw new Error("failed");
    45→  return r.json();
    46→}

   301→// ─── Open Drawer Dialog ─────────────────────────────────────────────────
   302→function OpenDrawerDialog({
   303→  open, onOpenChange, onSuccess,
   304→}: {
   305→  open: boolean;
   306→  onOpenChange: (o: boolean) => void;
   307→  onSuccess: () => void;
   308→}) {
   309→  const [opening, setOpening] = React.useState("");
   310→  const [userName, setUserName] = React.useState("");
   311→
   312→  const canOpen = !!opening && (parseFloat(opening) || 0) >= 0;
   313→
   314→  const mut = useMutation({
   315→    mutationFn: async () => {
   316→      const r = await fetch("/api/meat/cash/sessions", {
   317→        method: "POST",
   318→        headers: { "Content-Type": "application/json" },
   319→        body: JSON.stringify({
   320→          action: "OPEN",
   321→          opening: Number(opening) || 0,
   322→          userName: userName.trim() || undefined,
   323→        }),
   324→      });
   325→      if (!r.ok) {
   326→        const j = await r.json().catch(() => null);
   327→        throw new Error(j?.error || "failed");
   328→      }
   329→      return r.json();
   330→    },
   331→    onSuccess: () => {
   332→      toast.success("Cash drawer opened");
   333→      setOpening("");
   334→      setUserName("");
   335→      onSuccess();
   336→    },
   337→    onError: (e: Error) => toast.error(e.message || "Could not open drawer"),
   338→  });
   339→
   340→  return (
   341→    <Dialog open={open} onOpenChange={onOpenChange}>
   342→      <DialogContent className="max-w-md">
   343→        <DialogHeader>
   344→          <DialogTitle>Open Cash Drawer</DialogTitle>
   345→        </DialogHeader>
   346→        <div className="space-y-3 py-2">
   347→          <div>
   348→            <Label className="text-xs">Opening amount (Br)</Label>
   349→            <Input
   350→              value={opening}
   351→              onChange={(e) => setOpening(e.target.value)}
   352→              inputMode="decimal"
   353→              placeholder="0.00"
   354→              className="mt-1 tnum text-lg"
   355→              autoFocus
   356→            />
   357→            <p className="mt-1 text-[11px] text-muted-foreground">
   358→              Count the cash currently in the drawer and enter the total.
   359→            </p>
   360→          </div>
   361→          <div>
   362→            <Label className="text-xs">Opened by (optional)</Label>
   363→            <Input
   364→              value={userName}
   365→              onChange={(e) => setUserName(e.target.value)}
   366→              placeholder="Cashier name"
   367→              className="mt-1"
   368→            />
   369→          </div>
   370→        </div>
   371→        <DialogFooter>
   372→          <Button variant="outline" onClick={() => onOpenChange(false)}>Cancel</Button>
   373→          <Button
   374→            onClick={() => mut.mutate()}
   375→            disabled={mut.isPending || !canOpen}
   376→            className="bg-primary text-primary-foreground"
   377→          >
   378→            {mut.isPending ? "Opening…" : "Open Drawer"}
   379→          </Button>
   380→        </DialogFooter>
   381→      </DialogContent>
   382→    </Dialog>
   383→  );
   384→}

   386→// ─── Close Drawer Dialog ────────────────────────────────────────────────
   387→function CloseDrawerDialog({
   388→  open, onOpenChange, expected, onSuccess,
   389→}: {
   390→  open: boolean;
   391→  onOpenChange: (o: boolean) => void;
   392→  expected: number;
   393→  onSuccess: (session: CashSession) => void;
   394→}) {
   395→  const qc = useQueryClient();
   396→  const [counted, setCounted] = React.useState("");
   397→  const [note, setNote] = React.useState("");
   398→  const [userName, setUserName] = React.useState("");
   399→  const [transferEnabled, setTransferEnabled] = React.useState(false);
   400→  const [transferAmount, setTransferAmount] = React.useState("");
   401→  const [transferTo, setTransferTo] = React.useState<"MOBILE" | "BANK">("BANK");
   402→  const [transferDetail, setTransferDetail] = React.useState("");
   403→
   404→  const countedN = parseFloat(counted) || 0;
   405→  const diff = counted ? countedN - expected : 0;
   406→  const transferN = parseFloat(transferAmount) || 0;
   407→
   408→  const mut = useMutation({
   409→    mutationFn: async () => {
   410→      // If transfer is enabled, create a money move (OUT from CASH, IN to selected account)
   411→      if (transferEnabled && transferN > 0) {
   412→        await fetch("/api/meat/money", {
   413→          method: "POST",
   414→          headers: { "Content-Type": "application/json" },
   415→          body: JSON.stringify({
   416→            direction: "OUT", account: "CASH", amount: transferN,
   417→            reason: `Cash drawer transfer to ${transferTo}`,
   418→            userName: userName.trim() || "Abebe Owner",
   419→          }),
   420→        });
   421→        await fetch("/api/meat/money", {
   422→          method: "POST",
   423→          headers: { "Content-Type": "application/json" },
   424→          body: JSON.stringify({
   425→            direction: "IN", account: transferTo, detail: transferDetail || undefined,
   426→            amount: transferN,
   427→            reason: `Cash drawer transfer`,
   428→            userName: userName.trim() || "Abebe Owner",
   429→          }),
   430→        });
   431→      }
   432→      // Build note with transfer info if applicable
   433→      const transferNote = transferEnabled && transferN > 0
   434→        ? `Transferred ${formatBirr(transferN)} to ${transferTo}${transferDetail ? ` · ${transferDetail}` : ""}`
   435→        : "";
   436→      const fullNote = [note.trim(), transferNote].filter(Boolean).join(" | ") || undefined;
   437→      // Close the drawer
   438→      const r = await fetch("/api/meat/cash/sessions", {
   439→        method: "POST",
   440→        headers: { "Content-Type": "application/json" },
   441→        body: JSON.stringify({
   442→          action: "CLOSE",
   443→          counted: countedN,
   444→          note: fullNote,
   445→          userName: userName.trim() || undefined,
   446→        }),
   447→      });
   448→      if (!r.ok) {
   449→        const j = await r.json().catch(() => null);
   450→        throw new Error(j?.error || "failed");
   451→      }
   452→      return r.json();
   453→    },
   454→    onSuccess: (data) => {
   455→      toast.success(diff === 0 ? "Drawer closed — balanced" : diff > 0 ? `Drawer closed — over by ${formatBirr(diff)}` : `Drawer closed — short by ${formatBirr(Math.abs(diff))}`);
   456→      if (transferEnabled && transferN > 0) {
   457→        toast.success(`${formatBirr(transferN)} transferred to ${transferTo}`);
   458→      }
   459→      qc.invalidateQueries({ queryKey: ["dashboard"] });
   460→      qc.invalidateQueries({ queryKey: ["money"] });
   461→      qc.invalidateQueries({ queryKey: ["cash-sessions"] });
   462→      const session: CashSession = data.session;
   463→      setCounted(""); setNote(""); setUserName("");
   464→      setTransferEnabled(false); setTransferAmount(""); setTransferDetail("");
   465→      onSuccess(session);
   466→    },
   467→    onError: (e: Error) => toast.error(e.message || "Could not close drawer"),
   468→  });
   469→
   470→  return (
   471→    <Dialog open={open} onOpenChange={onOpenChange}>
   472→      <DialogContent className="max-h-[90vh] max-w-md overflow-y-auto">
   473→        <DialogHeader>
   474→          <DialogTitle>Close Cash Drawer</DialogTitle>
   475→          <DialogDescription className="sr-only">Count cash, optionally transfer to bank or mobile, then close the drawer.</DialogDescription>
   476→        </DialogHeader>
   477→        <div className="space-y-3 py-2">
   478→          {/* Expected */}
   479→          <div className="flex items-center justify-between rounded-lg bg-muted/40 px-3 py-2.5">
   480→            <span className="text-xs font-medium text-muted-foreground">Expected cash</span>
   481→            <span className="text-base font-bold tnum">{formatBirr(expected)}</span>
   482→          </div>
   483→
   484→          <div>
   485→            <Label className="text-xs">Counted total (Br)</Label>
   486→            <Input
   487→              value={counted}
   488→              onChange={(e) => setCounted(e.target.value)}
   489→              inputMode="decimal"
   490→              placeholder="0.00"
   491→              className="mt-1 tnum text-lg"
   492→              autoFocus
   493→            />
   494→          </div>
   495→
   496→          {/* Difference live */}
   497→          {counted ? (
   498→            <div className={cn(
   499→              "flex items-center justify-between rounded-lg px-3 py-2.5 ring-1",
   500→              diff === 0
   501→                ? "bg-emerald-500/10 ring-emerald-500/25"
   502→                : diff > 0
   503→                ? "bg-emerald-500/10 ring-emerald-500/25"
   504→                : "bg-red-500/10 ring-red-500/25",
   505→            )}>
   506→              <span className="text-xs font-medium text-muted-foreground">Difference</span>
   507→              <Pill tone={diffTone(diff)}>{diffLabel(diff)}</Pill>
   508→            </div>
   509→          ) : (
   510→            <div className="flex items-center justify-between rounded-lg bg-muted/30 px-3 py-2.5">
   511→              <span className="text-xs font-medium text-muted-foreground">Difference</span>
   512→              <span className="text-sm font-semibold text-muted-foreground tnum">—</span>
   513→            </div>
   514→          )}
   515→
   516→          {/* Transfer option */}
   517→          <div className="rounded-lg border border-border/60 p-2.5">
   518→            <label className="flex items-center gap-2">
   519→              <input type="checkbox" checked={transferEnabled} onChange={(e) => setTransferEnabled(e.target.checked)} className="h-4 w-4 rounded accent-primary" />
   520→              <span className="text-xs font-semibold">Transfer cash to Bank or Mobile</span>
   521→            </label>
   522→            {transferEnabled && (
   523→              <div className="mt-2.5 space-y-2">
   524→                <div className="grid grid-cols-2 gap-1.5">
   525→                  <button type="button" onClick={() => { setTransferTo("BANK"); setTransferDetail(""); }}
   526→                    className={cn("rounded-lg py-2 text-xs font-semibold", transferTo === "BANK" ? "bg-primary text-primary-foreground" : "bg-muted/60 text-muted-foreground")}>
   527→                    🏦 Bank
   528→                  </button>
   529→                  <button type="button" onClick={() => { setTransferTo("MOBILE"); setTransferDetail(""); }}
   530→                    className={cn("rounded-lg py-2 text-xs font-semibold", transferTo === "MOBILE" ? "bg-primary text-primary-foreground" : "bg-muted/60 text-muted-foreground")}>
   531→                    📱 Mobile
   532→                  </button>
   533→                </div>
   534→                <div>
   535→                  <Label className="text-[10px]">Amount to transfer (Br)</Label>
   536→                  <Input value={transferAmount} onChange={(e) => setTransferAmount(e.target.value)} inputMode="decimal" placeholder="e.g. 20000" className="mt-0.5 h-9 tnum text-sm" />
   537→                  <div className="mt-1 flex gap-1.5">
   538→                    <button type="button" onClick={() => setTransferAmount(String(countedN || expected))} className="rounded-md bg-muted/60 px-2 py-1 text-[10px] font-semibold">Full amount</button>
   539→                    <button type="button" onClick={() => setTransferAmount(String(Math.round((countedN || expected) / 2)))} className="rounded-md bg-muted/60 px-2 py-1 text-[10px] font-semibold">Half</button>
   540→                  </div>
   541→                </div>
   542→                <div>
   543→                  <Label className="text-[10px]">{transferTo === "MOBILE" ? "Provider" : "Bank"}</Label>
   544→                  <div className="mt-0.5">
   545→                    <AccountProviderSelect method={transferTo} value={transferDetail} onChange={setTransferDetail} />
   546→                  </div>
   547→                </div>
   548→              </div>
   549→            )}
   550→          </div>
   551→
   552→          <div>
   553→            <Label className="text-xs">Closed by (optional)</Label>
   554→            <Input value={userName} onChange={(e) => setUserName(e.target.value)} placeholder="Cashier name" className="mt-1" />
   555→          </div>
   556→
   557→          <div>
   558→            <Label className="text-xs">Note (optional)</Label>
   559→            <Textarea value={note} onChange={(e) => setNote(e.target.value)} rows={2} placeholder="e.g. shift handover notes" className="mt-1" />
   560→          </div>
   561→        </div>
   562→
   563→        <DialogFooter>
   564→          <Button variant="outline" onClick={() => onOpenChange(false)}>Cancel</Button>
   565→          <Button
   566→            onClick={() => mut.mutate()}
   567→            disabled={mut.isPending || !counted}
   568→            className="bg-primary text-primary-foreground"
   569→          >
   570→            {mut.isPending ? "Closing…" : "Close Cash Drawer"}
   571→          </Button>
   572→        </DialogFooter>
   573→      </DialogContent>
   574→    </Dialog>
   575→  );
   576→}

   578→// ─── Helpers ────────────────────────────────────────────────────────────
   579→function diffTone(diff: number): "good" | "bad" {
   580→  if (diff === 0) return "good";
   581→  return diff > 0 ? "good" : "bad";
   582→}
   583→
   584→function diffAccent(diff: number): "emerald" | "red" | undefined {
   585→  if (diff === 0) return "emerald";
   586→  return diff > 0 ? "emerald" : "red";
   587→}
   588→
   589→function diffLabel(diff: number): string {
   590→  if (diff === 0) return "Balanced";
   591→  if (diff > 0) return `Over by ${formatBirr(diff)}`;
   592→  return `Short by ${formatBirr(Math.abs(diff))}`;
   593→}

   595→// ─── Small display pieces ───────────────────────────────────────────────
   596→function MetaCell({
   597→  label, value, tnum, tone, accent,
   598→}: {
   599→  label: string;
   600→  value: string;
   601→  tnum?: boolean;
   602→  tone?: "good" | "bad";
   603→  accent?: "emerald" | "red";
   604→}) {
   605→  const valueCls = tone === "good"
   606→    ? "text-emerald-400"
   607→    : tone === "bad"
   608→    ? "text-red-400"
   609→    : accent === "emerald"
   610→    ? "text-emerald-400"
   611→    : accent === "red"
   612→    ? "text-red-400"
   613→    : "text-foreground";
   614→  return (
   615→    <div className="rounded-lg bg-muted/40 px-2.5 py-2">
   616→      <p className="text-[10px] uppercase tracking-wider text-muted-foreground">{label}</p>
   617→      <p className={cn("truncate text-xs font-medium", valueCls, tnum && "tnum")}>{value}</p>
   618→    </div>
   619→  );
   620→}
